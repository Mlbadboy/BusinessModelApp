logging.basicConfig(level=logging.INFO,

class TradingBot:
    def __init__(self, symbol='BTC/USDT'):
        self.symbol = symbol.replace('/', '')  # Binance format
        self.exchange = ccxt.binance({'enableRateLimit': True})
        self.model = None
        self.scaler = StandardScaler()
        self.feature_columns = None
        self.initialize_model()
        self.db_conn = sqlite3.connect("trading_bot.db", check_same_thread=False)
        self.db_cursor = self.db_conn.cursor()
        self.create_db_table()
        self.start_websocket()
    
    def initialize_model(self):
        """Load or train the model"""
        try:
            self.model = joblib.load('trading_model.pkl')
            self.scaler = joblib.load('scaler.pkl')
            logging.info("Loaded existing model successfully")
        except (FileNotFoundError, ValueError, TypeError) as e:
            logging.warning(f"Model loading failed: {str(e)}. Training new model...")
            self.train_model()
            self.save_model()
    
    def create_db_table(self):
        """Create trade logging table"""
        self.db_cursor.execute('''CREATE TABLE IF NOT EXISTS trades 
                                  (timestamp TEXT, price REAL, trade_type TEXT, confidence REAL)''')
        self.db_conn.commit()
    
    def start_websocket(self):
        """Start Binance WebSocket for real-time price updates"""
        twm = ThreadedWebsocketManager()
        twm.start()
        twm.start_symbol_ticker_socket(callback=self.handle_socket_message, symbol=self.symbol.lower())
    
    def handle_socket_message(self, msg):
        """Handle incoming WebSocket messages"""
        logging.info(f"Live Price Update: {msg['p']}")
    
    def fetch_realtime_data(self):
        """Fetch real-time data from Binance"""
        try:
            data = self.exchange.fetch_ohlcv(self.symbol, '1m', limit=100)
            if not data:
                logging.error("No data returned from exchange.")
                return pd.DataFrame()
            
            df = pd.DataFrame(data, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
            df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
            df.set_index('timestamp', inplace=True)
            return df
        except Exception as e:
            logging.error(f"Error fetching real-time data: {str(e)}")
            return pd.DataFrame()
    
    def analyze_market(self):
        """Perform real-time analysis"""
        try:
            df = self.fetch_realtime_data()
            if df.empty:
                logging.error("No real-time data fetched.")
                return None
            
            df['RSI'] = talib.RSI(df['close'], timeperiod=14)
            df['MACD'], _, _ = talib.MACD(df['close'])
            df['ATR'] = talib.ATR(df['high'], df['low'], df['close'], timeperiod=14)
            df['SMA_50'] = talib.SMA(df['close'], timeperiod=50)
            df['SMA_200'] = talib.SMA(df['close'], timeperiod=200)
            df['VWAP'] = (df['close'] * df['volume']).cumsum() / df['volume'].cumsum()
            
            market_trend = "Upwards" if df['SMA_50'].iloc[-1] > df['SMA_200'].iloc[-1] else "Downwards"
            sentiment_score = self.get_sentiment_score()
            sentiment = "Bullish" if sentiment_score > 0 else "Bearish"
            
            trade_action = "Hold"
            if df['RSI'].iloc[-1] < 30:
                trade_action = "Buy"
            elif df['RSI'].iloc[-1] > 70:
                trade_action = "Sell"
            
            atr_value = df['ATR'].iloc[-1]
            analysis = {
                'timestamp': datetime.now().isoformat(),
                'current_price': df['close'].iloc[-1],
                'market_trend': market_trend,
                'execute_trade_price': df['close'].iloc[-1] * 0.995 if trade_action == "Buy" else df['close'].iloc[-1] * 1.005,
                'trade_type': trade_action,
                'target': f"{atr_value * 2.5:.2f} points",
                'stop_loss': f"{atr_value * 1.5:.2f} points",
                'volatility': f"{atr_value:.2f}%",
                'sentiment': sentiment
            }
            
            self.log_trade(analysis)
            logging.info(analysis)
            return analysis
        except Exception as e:
            logging.error(f"Analysis error: {str(e)}", exc_info=True)
            return None
    
    def log_trade(self, analysis):
        """Log trade into SQLite database"""
        self.db_cursor.execute("INSERT INTO trades VALUES (?, ?, ?, ?)", 
                               (analysis['timestamp'], analysis['current_price'], analysis['trade_type'], 0.95))
        self.db_conn.commit()
    
    def get_sentiment_score(self):
        """Fetch news sentiment score"""
        try:
            news = requests.get("https://newsapi.org/v2/everything?q=bitcoin&apiKey=YOUR_API_KEY").json()
            headlines = " ".join([article["title"] for article in news["articles"]])
            return TextBlob(headlines).sentiment.polarity
        except:
            return 0
